# 9. 
#include &lt;stdio.h&gt; 앞에 '#define _CRT_SECURE_NO_WARNINGS' 혹은 '#pragma warning(disable: 4996)'를 추가한다.

# 10. 
```C
#pragma warning(disable : 4996)
#include <stdio.h>

int main()
{
    char str[100];

    printf("문자열 입력: ");
    scanf("%s", str);

    printf("입력한 문자열: %s", str);

    return 0;
}
```
